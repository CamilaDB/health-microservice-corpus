<EXAMPLE>

<INPUT>

<METHOD_SOURCE>
async create(dto: CreateUserDto) {
const existing = await repository.findOne({
email: dto.email,
});

if (existing) {
throw new ConflictException();
}

const entity = repository.create(dto);

return repository.save(entity);
}
</METHOD_SOURCE>

</INPUT>

<OUTPUT>

<TEST_OUTPUT>
describe('create', () => {
it('should throw ConflictException when user already exists', async () => {
repositoryMock.findOne.mockResolvedValueOnce({
id: '1',
email: '[test@test.com](mailto:test@test.com)',
});

```
await expect(
  service.create({
    email: 'test@test.com',
  } as CreateUserDto),
).rejects.toThrow(ConflictException);

expect(repositoryMock.create).not.toHaveBeenCalled();
expect(repositoryMock.save).not.toHaveBeenCalled();
```

});

it('should save created entity', async () => {
repositoryMock.findOne.mockResolvedValueOnce(null);

```
const createdEntity = {
  email: 'test@test.com',
};

repositoryMock.create.mockReturnValueOnce(createdEntity);

repositoryMock.save.mockResolvedValueOnce({
  id: '1',
  ...createdEntity,
});

await service.create({
  email: 'test@test.com',
} as CreateUserDto);

expect(repositoryMock.create).toHaveBeenCalledWith({
  email: 'test@test.com',
});

expect(repositoryMock.save).toHaveBeenCalledWith(createdEntity);
```

});
});
</TEST_OUTPUT>

</OUTPUT>

</EXAMPLE>
