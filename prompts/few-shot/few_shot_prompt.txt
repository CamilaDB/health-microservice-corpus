async create(dto: CreateUserDto) {
  const existing = await this.userRepository.findOne({
    email: dto.email,
  });

  if (existing) {
    throw new ConflictException();
  }

  const entity = this.userRepository.create(dto);

  return this.userRepository.save(entity);
}
</METHOD_SOURCE>

describe('create', () => {
  it('should throw ConflictException when user already exists', async () => {
    userRepositoryMock.findOne.mockResolvedValueOnce({
      id: '1',
      email: 'test@test.com',
    });

    await expect(
      service.create({ email: 'test@test.com' } as CreateUserDto),
    ).rejects.toThrow(ConflictException);

    expect(userRepositoryMock.create).not.toHaveBeenCalled();
    expect(userRepositoryMock.save).not.toHaveBeenCalled();
  });

  it('should save and return the created entity', async () => {
    userRepositoryMock.findOne.mockResolvedValueOnce(null);

    const createdEntity = { email: 'test@test.com' };
    userRepositoryMock.create.mockReturnValueOnce(createdEntity);
    userRepositoryMock.save.mockResolvedValueOnce({ id: '1', ...createdEntity });

    await service.create({ email: 'test@test.com' } as CreateUserDto);

    expect(userRepositoryMock.create).toHaveBeenCalledWith({ email: 'test@test.com' });
    expect(userRepositoryMock.save).toHaveBeenCalledWith(createdEntity);
  });
});
</TEST_OUTPUT>
</OUTPUT>

<EXAMPLE_2>

<INPUT>
<METHOD_SOURCE>
async validate(id: string) {
  const user = await this.userRepository.findById(id);

  if (!user) {
    throw new NotFoundException();
  }

  if (!user.active) {
    throw new BadRequestException();
  }

  return user;
}
</METHOD_SOURCE>
</INPUT>

<OUTPUT>
<TEST_OUTPUT>
describe('validate', () => {
  it('should throw NotFoundException when user does not exist', async () => {
    userRepositoryMock.findById.mockResolvedValueOnce(null);

    await expect(service.validate('1')).rejects.toThrow(NotFoundException);
  });

  it('should throw BadRequestException when user is inactive', async () => {
    userRepositoryMock.findById.mockResolvedValueOnce({ id: '1', active: false });

    await expect(service.validate('1')).rejects.toThrow(BadRequestException);
  });

  it('should return the user when active', async () => {
    const user = { id: '1', active: true };
    userRepositoryMock.findById.mockResolvedValueOnce(user);

    const result = await service.validate('1');

    expect(result).toBe(user);
  });
});
</TEST_OUTPUT>
</OUTPUT>

</EXAMPLE_2>

<EXAMPLE_3>

<INPUT>
<METHOD_SOURCE>
getRole(user: User): UserRole {
  if (!user.roles || user.roles.length === 0) {
    throw new BadRequestException('No roles assigned');
  }

  return user.roles[0];
}
</METHOD_SOURCE>
</INPUT>

<OUTPUT>
<TEST_OUTPUT>
describe('getRole', () => {
  it('should throw BadRequestException when user has no roles', () => {
    expect(() =>
      service.getRole({ roles: [] } as User),
    ).toThrow(BadRequestException);
  });

  it('should return the first role when roles are present', () => {
    const result = service.getRole({ roles: [UserRole.ADMIN] } as User);

    expect(result).toBe(UserRole.ADMIN);
  });
});
</TEST_OUTPUT>
</OUTPUT>
